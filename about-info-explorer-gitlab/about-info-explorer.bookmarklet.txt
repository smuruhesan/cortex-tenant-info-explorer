  function parseTsMap(src, name) {
    if (!src) return {};

    const i = src.indexOf(name);
    if (i < 0) return out;
    const open = src.indexOf("{", i);
    if (open < 0) return out;

    for (let j = open; j < src.length; j++) {
      if (src[j] === "{") d++;
      else if (src[j] === "}") { d--; if (!d) { end = j; break; } }
    }
    if (end < 0) return out;
    for (const m of src.slice(open + 1, end).matchAll(/['"]?([A-Za-z_]\w*)['"]?\s*:\s*['"]([^'"]+)['"]/g)) {

    return out;
